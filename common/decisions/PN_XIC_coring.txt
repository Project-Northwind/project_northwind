XIC_chinese_state_integration = {
	XIC_integrate_north_basic = {
		icon = generic_prepare_civil_war
		allowed = { original_tag = XIC }
		state_target = yes
		targets = { 597 607 608 609 610 611 612 614 615 616 621 622 743 744 746 1032 1038 1039 1040 1043 1045 1046 }
		target_trigger = {
			FROM = { XIC_is_north_integration_state = yes }
		}
		on_map_mode = map_only
		visible = {
			OR = {
				XIC_integration_basic_tier = yes
				FROM = { has_state_flag = XIC_chinese_integration_basic_active }
			}
			FROM = {
				OR = {
					has_state_flag = XIC_chinese_integration_basic_active
					AND = {
						is_owned_by = ROOT
						NOT = { is_core_of = ROOT }
						NOT = { has_state_flag = XIC_chinese_integration_in_progress }
					}
				}
			}
		}
		available = {
			XIC_can_administer_chinese_states = yes
			XIC_integration_basic_tier = yes
			FROM = {
				XIC_chinese_state_ready_for_integration = yes
				NOT = { has_state_flag = XIC_chinese_integration_in_progress }
			}
		}
		cost = 50
		days_remove = 90
		fire_only_once = no
		days_re_enable = 1
		complete_effect = {
			hidden_effect = {
				FROM = {
					set_state_flag = XIC_chinese_integration_in_progress
					set_state_flag = XIC_chinese_integration_basic_active
				}
			}
		}
		cancel_trigger = {
			OR = {
				NOT = { XIC_can_administer_chinese_states = yes }
				FROM = { NOT = { XIC_chinese_state_ready_for_integration = yes } }
			}
		}
		cancel_effect = {
			hidden_effect = { FROM = { XIC_clear_chinese_integration_flags_effect = yes } }
		}
		remove_effect = {
			effect_tooltip = {
				FROM = {
					add_core_of = ROOT
					remove_claim_by = ROOT
				}
			}
			hidden_effect = { FROM = { XIC_complete_chinese_state_integration_effect = yes } }
		}
		ai_will_do = { factor = 1 }
	}
