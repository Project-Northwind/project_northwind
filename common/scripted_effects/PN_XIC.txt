		limit = { has_country_flag = XIC_puyi_no_offer }
		XIC_puyi_uniter_settlement_effect = yes
	}
	else_if = {
		limit = { XIC_puyi_available = yes }
		if = {
			limit = { has_country_flag = XIC_puyi_offer_citizenship }
			set_country_flag = XIC_republican_settlement_chosen
		}
		set_country_flag = XIC_puyi_letter_pending
		MAN = { country_event = { id = PN_XIC.35 } }
		country_event = { id = PN_XIC.39 days = 1 }
	}
	else = {
		XIC_puyi_uniter_settlement_effect = yes
		country_event = { id = PN_XIC.38 }
	}
}

			set_character_name = XIC_xuangtong_emperor_puyi

					large = GFX_portrait_man_henry_pu_yi_emperor

				}
			}
			add_advisor_role = {
				advisor = {
					slot = political_advisor
					idea_token = XIC_xuangtong_emperor_puyi
					cost = 0
					can_be_fired = no

				}
				activate = yes
			}
		}

		set_variable = { PN_secondary_leader_role = 1 }
		MAN_aisin_gioro_puyi = { PN_set_secondary_leader_effect = yes }
