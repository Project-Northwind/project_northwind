# Country scope. Chinese geography is independent of existing XIC cores.
XIC_claim_chinese_states_effect = {
	custom_effect_tooltip = XIC_chinese_state_claims_tt
	hidden_effect = {
		every_state = {
			limit = {
				XIC_is_chinese_state = yes
				NOT = { is_core_of = ROOT }
			}
			add_claim_by = ROOT
		}
	}
}

# State scope, ROOT is the integrating country. Recheck all conditions at expiry.
XIC_complete_chinese_state_integration_effect = {
	if = {
		limit = {
			has_state_flag = XIC_chinese_integration_in_progress
			XIC_chinese_state_ready_for_integration = yes
			ROOT = { XIC_can_administer_chinese_states = yes }
		}
		add_core_of = ROOT
		remove_claim_by = ROOT
	}
	hidden_effect = { XIC_clear_chinese_integration_flags_effect = yes }
}

XIC_postwar_settlement_effect = {
	custom_effect_tooltip = XIC_replace_shangdong_spirits_tt
	hidden_effect = {
		remove_dynamic_modifier = { modifier = XIC_industrial_reconstruction }
		if = {
			limit = { has_idea = XIC_national_reconstructionism_3 }

		}
		else_if = {
			limit = { has_idea = XIC_national_reconstructionism_2 }
			remove_ideas = XIC_national_reconstructionism_2
		}
		else_if = {
			limit = { has_idea = XIC_national_reconstructionism }
			remove_ideas = XIC_national_reconstructionism

		set_variable = { XIC_postwar_construction = -0.35 }
		set_variable = { XIC_postwar_consumer_goods = 0.25 }
		set_variable = { XIC_postwar_efficiency_growth = -0.20 }
		set_variable = { XIC_postwar_efficiency_cap = 0 }
		set_variable = { XIC_postwar_resources = 0 }
		add_dynamic_modifier = { modifier = XIC_fallout_from_the_war }
		add_ideas = XIC_national_legitimacy
	}
	custom_effect_tooltip = generic_skip_one_line_tt
	custom_effect_tooltip = XIC_postwar_economy_tt
	custom_effect_tooltip = generic_skip_one_line_tt
	custom_effect_tooltip = XIC_national_legitimacy_start_tt
}
