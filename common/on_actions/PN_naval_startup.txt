# Project Northwind naval startup; also works in all-AI/observer games.
on_actions = {
	on_startup = {
		effect = {
			set_global_flag = vnr_enabled
			set_global_flag = navy_rework_english
			every_country = {
				limit = {
					is_ai = no
				}
				set_country_flag = navy_rework_splash_show
				set_variable = {
					navy_rework_welcome_splash_variable = 1
				}
			}
			set_variable = {
				vnr_ai_difficulty = 0
			}
			set_variable = {
				vnr_ai_production = 0
			}
			add_to_array = {
				global.vnr_naval_powers = ENG
			}
			add_to_array = {
				global.vnr_naval_powers = FRA
			}
			add_to_array = {
				global.vnr_naval_powers = GER
			}
			add_to_array = {
				global.vnr_naval_powers = ITA
			}
			add_to_array = {
				global.vnr_naval_powers = JAP
			}
			add_to_array = {
				global.vnr_naval_powers = SOV
			}
			add_to_array = {
				global.vnr_naval_powers = USA
			}
			set_global_flag = {
				flag = MTG_naval_treaty_violator
				value = 0
			}
			random_country = {
				limit = {
					is_ai = no
				}
				PN_initialize_navy = yes
			}
			if = {
				limit = {
					NOT = {
						has_global_flag = PN_naval_initialized
					}
				}
				random_country = {
					PN_initialize_navy = yes
				}
			}
		}
	}
}
