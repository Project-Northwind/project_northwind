# Country scope. Formation flags keep existing saves eligible for integration.
XIC_can_administer_chinese_states = {
	XIC_campaign_government_established = yes
	is_subject = no
	has_capitulated = no
}

# One completed administrative focus provides local offices; both provide a
# professional administration. Each branch has its own real focus gates.
XIC_has_local_integration_administration = {
	OR = {
		AND = {
			has_country_flag = XIC_national_peoples_republic_formed
			has_completed_focus = PN_establish_territorial_league_branches
		}
		AND = {
			has_country_flag = XIC_national_state_formed
			OR = {
				has_completed_focus = PN_spread_the_ideology_of_reconstructionism
				has_completed_focus = PN_the_people_only_need_me
			}
		}
	}
}

XIC_has_professional_integration_administration = {
	OR = {
		AND = {
			has_country_flag = XIC_national_peoples_republic_formed
			has_completed_focus = PN_establish_territorial_league_branches
			has_completed_focus = PN_a_state_beyond_the_officer_corps
		}
		AND = {
			has_country_flag = XIC_national_state_formed
			has_completed_focus = PN_spread_the_ideology_of_reconstructionism
			has_completed_focus = PN_the_people_only_need_me
		}
	}
}

XIC_integration_basic_tier = {
	NOT = { XIC_has_local_integration_administration = yes }
}

XIC_integration_local_tier = {
	XIC_has_local_integration_administration = yes
	NOT = { XIC_has_professional_integration_administration = yes }
}

XIC_integration_professional_tier = {
	XIC_has_professional_integration_administration = yes
}

# State scope, ROOT is the integrating country. KX's ordinary provincial
# integration uses ownership and full control, without compliance or supply gates.

	XIC_is_chinese_state = yes
	is_owned_by = ROOT
	is_fully_controlled_by = ROOT
	NOT = { is_core_of = ROOT }
}
