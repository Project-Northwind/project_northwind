country_event = {
	id = PN_XIC_campaigns.1
	title = PN_XIC_campaigns.1.t
	desc = PN_XIC_campaigns.1.desc
	picture = GFX_report_event_chinese_officers
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		exists = yes
		is_subject = no
		is_in_faction = no
		NOT = { has_war_with = XIC }
		XIC = { exists = yes XIC_campaign_government_established = yes }
		XIC = { has_country_flag = XIC_national_peoples_republic_formed }
	}

	option = {
		name = PN_XIC_campaigns.1.a
		ai_chance = {
			base = 35
			modifier = {
				factor = 2
				XIC = { has_country_flag = XIC_civilian_provincial_service }
			}
			modifier = {
				factor = 1.5
				XIC = { has_country_flag = XIC_direct_integration_priority }
			}
		}
		custom_effect_tooltip = XIC_provincial_integration_completed_tt
		if = {
			limit = {
				hidden_trigger = {
					exists = yes
					is_subject = no
					is_in_faction = no
					NOT = {
						has_war_with = XIC
					}
				}
			}
			hidden_effect = {
				XIC_transfer_provincial_officers_effect = yes
			}
			XIC = {
				hidden_effect = {
					annex_country = {
						target = ROOT
						transfer_troops = yes
					}
				}
				country_event = { id = PN_XIC_campaigns.4 days = 1 }
			}
		}
		else = {
			hidden_effect = {
				set_country_flag = XIC_integration_refused
			}
		}
	}
	option = {
		name = PN_XIC_campaigns.1.b
		ai_chance = {
			base = 40
			modifier = {
				factor = 2
				XIC = {
					OR = {
						has_country_flag = XIC_officer_provincial_service
						has_country_flag = XIC_governors_local_compacts
					}
				}
			}
			modifier = {
				factor = 1.5
				XIC = { has_country_flag = XIC_provincial_autonomy_priority }
			}
		}
		custom_effect_tooltip = XIC_provincial_autonomy_tt
		hidden_effect = {
			if = {
				limit = { exists = yes is_subject = no is_in_faction = no NOT = { has_war_with = XIC } }
				set_country_flag = XIC_provincial_autonomy_accepted
				XIC = { puppet = ROOT }
			}
			else = {
				set_country_flag = XIC_integration_refused
			}
		}
	}
	option = {
		name = PN_XIC_campaigns.1.c
		ai_chance = { base = 25 }
		hidden_effect = {
			set_country_flag = XIC_integration_refused
		}
		custom_effect_tooltip = XIC_provincial_refusal_tt
	}
}
