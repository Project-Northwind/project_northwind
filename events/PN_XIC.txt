country_event = {
	id = PN_XIC.17
	title = PN_XIC.17.t
	desc = PN_XIC.17.desc
	picture = GFX_report_event_chinese_officers
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		tag = PRC
		XIC = {
			exists = yes
			has_country_flag = XIC_envoy_to_wang_ming_sent
			NOT = { has_country_flag = XIC_national_settlement_resolved }
		}
	}

	option = {
		name = PN_XIC.17.a
		trigger = {
			XIC_wang_ming_available = yes
			XIC = {
				has_country_flag = XIC_envoy_to_wang_ming_sent
				NOT = { has_country_flag = XIC_national_settlement_resolved }
			}
		}
		ai_chance = { base = 60 }
		custom_effect_tooltip = XIC_accept_settlement_tt
		if = {
			limit = {
				XIC_wang_ming_available = yes
				XIC = {
					has_country_flag = XIC_envoy_to_wang_ming_sent
					NOT = { has_country_flag = XIC_national_settlement_resolved }
				}
			}
			XIC = {
				set_country_flag = XIC_communist_settlement_accepted
				complete_national_focus = PN_the_national_peoples_republic_of_china
				country_event = { id = PN_XIC.18 }
			}
		}
		else = {
			if = {
				limit = {
					XIC = {
						has_country_flag = XIC_envoy_to_wang_ming_sent
						NOT = { has_country_flag = XIC_national_settlement_resolved }
					}
				}
				XIC = {
					XIC_choose_national_state_effect = yes
					country_event = { id = PN_XIC.20 }
				}
			}
		}
	}
	option = {
		name = PN_XIC.17.b
		ai_chance = { base = 40 }
		custom_effect_tooltip = XIC_reject_settlement_tt
		if = {
			limit = {
				XIC = {
					has_country_flag = XIC_envoy_to_wang_ming_sent
					NOT = { has_country_flag = XIC_national_settlement_resolved }
				}
			}
			XIC = {
				XIC_choose_national_state_effect = yes
				country_event = { id = PN_XIC.19 }
			}
		}
	}
}
