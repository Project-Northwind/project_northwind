	fire_only_once = yes

	trigger = {
		original_tag = XIC
	}

	option = {
		name = PN_XIC_programmes.5.a
		ai_chance = { base = 50 }
		trigger = {
			owns_state = 1035
			has_full_control_of_state = 1035
		}
		if = {
			limit = { owns_state = 1035 has_full_control_of_state = 1035 }
			add_equipment_to_stockpile = { type = infantry_equipment_1 amount = 500 producer = CHI }
			add_equipment_to_stockpile = { type = artillery_equipment_1 amount = 40 producer = CHI }
		}
		else = { add_political_power = 25 }
	}

	option = {
		name = PN_XIC_programmes.5.b
		ai_chance = { base = 50 }
		trigger = {
			has_political_power > 74
			owns_state = 1035
			has_full_control_of_state = 1035
			1035 = { free_building_slots = { building = arms_factory size > 0 include_locked = no } }
		}
		if = {
			limit = {
				has_political_power > 74
				owns_state = 1035
				has_full_control_of_state = 1035
				1035 = { free_building_slots = { building = arms_factory size > 0 include_locked = no } }
			}
			add_political_power = -75
			1035 = { add_building_construction = { type = arms_factory level = 1 instant_build = yes } }
		}
		else = { add_political_power = 25 }
	}

	option = {
		name = PN_XIC_programmes.5.c
		ai_chance = { base = 50 }
		trigger = {
			OR = { NOT = { owns_state = 1035 } NOT = { has_full_control_of_state = 1035 } }
		}
		add_political_power = 25
	}
}
